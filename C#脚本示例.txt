	{        // 生成112pin位置度的实际基准值
        private static List<double> GenerateActual112PinPositionValues()
        {
            List<double> actualValues = new List<double>();
            
            // 112pin参数：
            // 台阶区域1：pin28-29之间
            // 台阶区域2：pin56-57之间 
            // 台阶区域3：pin84-85之间（基准位置）
            // pin28到基准距离：43.315
            // pin56到基准距离：22.555  
            // pin84到基准距离：2.005
            // pin112到基准距离：18.205
            // pin间距：0.6（除台阶区域外）
            
            // 计算pin1到pin28的距离
            for (int i = 1; i <= 28; i++)
            {
                double distance = 14.715 + (28 - i) * 0.6;
                actualValues.Add(distance);
            }
            
            // 计算pin29到pin42的距离（跳过台阶区域1）
            for (int i = 29; i <= 42; i++)
            {
                double distance = 2.005 + (42 - i) * 0.6;
                actualValues.Add(distance);
            }
            
            // 计算pin43到pin70的距离（跳过台阶区域2）
            for (int i = 43; i <= 70; i++)
            {
                double distance = 2.005 + (i - 43) * 0.6;
                actualValues.Add(distance);
            }
            

            
            return actualValues;
        }
        
		public static void Execute(Guid processkey)
		{
			var Script = new Script_Function(processkey);
            try
            {

                Script.SetVarValue(2, "status", 1);
                Script.SetVarValue(2, "status1", 1);
                double gap_threshold_up=(double)Script.GetGlobalVarValue("gap上限");
                double gap_threshold_down=(double)Script.GetGlobalVarValue("gap下限");            
                double position_threshold=(double)Script.GetGlobalVarValue("位置度");
                double angle_threshold=(double)Script.GetGlobalVarValue("pin歪斜角度");
                double compensation_A_threshold=(double)Script.GetGlobalVarValue("焊接面A位置度补偿");
                double compensation_B_threshold=(double)Script.GetGlobalVarValue("焊接面B位置度补偿");
                double deform=(double)Script.GetGlobalVarValue("pin歪斜距离");
                //实际pin位置
                List<double> actualValues = GenerateActual112PinPositionValues();
                double nindex = (double)Script.GetVarValue(2, "index");
                int index=(int)nindex;
                double gap = Math.Round((double)Script.GetValue(67,"Distance"),3);
                List<double>gap_list = (List<double>)Script.GetVarValue(2,"gap_list");
                gap_list.Add(gap);
                //检测项1：gap
                if (gap < gap_threshold_down || gap > gap_threshold_up)
                {       //MessageBox.Show("3");             
                        Script.SetVarValue(2, "message", "gap检查NG,gap:"+gap+",序号:" + (index + 1).ToString());
                        Script.SetVarValue(2,"front_result", false);
                        Script.SetVarValue(2, "status", 0);
                        Script.SetVarValue(1,"num1",0);
                }
                //检测项2：位置度（top）
                List<double> topPositionDifferences = (List<double>)(Script.GetGlobalVarValue("top_position_differences"));                          
                double top_location = Math.Round((double)Script.GetValue(52,"MeanDistance"),3);
                List<double>list_top_location = (List<double>)Script.GetVarValue(2,"top_location");
                list_top_location.Add(top_location);
                double top_difference = Math.Abs(top_location - actualValues[70-index])+compensation_B_threshold;
                topPositionDifferences.Add(top_difference);
                if (top_difference > position_threshold)
                { 
                      Script.SetVarValue(2, "status", 0);
                      Script.SetVarValue(2,"front_result", false);
                      Script.SetVarValue(1,"num2",0);
                }

                List<double> bottomPositionDifferences = (List<double>)(Script.GetGlobalVarValue("bottom_position_differences"));
                double bottom_location = Math.Round((double)Script.GetValue(58,"MeanDistance"),3);
                List<double>list_bottom_location = (List<double>)Script.GetVarValue(2,"bottom_location");
                list_bottom_location.Add(bottom_location);
                double bottom_difference = Math.Abs(bottom_location - actualValues[70-index])+compensation_A_threshold;
                bottomPositionDifferences.Add(bottom_difference);
                if (bottom_difference > position_threshold)
                {
                      Script.SetVarValue(2, "status1", 0);
                      Script.SetVarValue(2,"front_result", false);
                      Script.SetVarValue(1,"num2",0);
                }

                List<Point2D> bdata = (List<Point2D>)(Script.GetValue(51, "point2ds"));
                List<double> Bdata = (List<double>)(Script.GetVarValue(2, "Bpin针数据"));
                double adata = (double)Script.GetValue(58,"MeanDistance");
